    const inputType = document.getElementById('inputType');
const fieldsContainer = document.getElementById('fieldsContainer');
const dynamicForm = document.getElementById('dynamicForm');
const resetBtn = document.getElementById('resetBtn');
const result = document.getElementById('result');

let fieldCount = 0;


// Add field when dropdown changes
inputType.addEventListener('change', () => {

  const type = inputType.value;

  if (!type) {
    return;
  }

  fieldCount++;

  const fieldWrapper = document.createElement('div');
  fieldWrapper.classList.add('field');

  const fieldName = `${type}${fieldCount}`;

  // Checkbox
  if (type === 'checkbox') {

    fieldWrapper.classList.add('checkbox-field');

    const input = document.createElement('input');

    input.type = 'checkbox';
    input.name = fieldName;
    input.id = fieldName;

    const label = document.createElement('label');

    label.htmlFor = fieldName;
    label.textContent = `Checkbox ${fieldCount}`;

    fieldWrapper.append(input, label);

  }

  // Other input types
  else {

    const label = document.createElement('label');

    label.htmlFor = fieldName;
    label.textContent = `${type} ${fieldCount}`;

    const input = document.createElement('input');

    input.type = type;
    input.name = fieldName;
    input.id = fieldName;

    fieldWrapper.append(label, input);
  }


  fieldsContainer.appendChild(fieldWrapper);

  // Reset dropdown
  inputType.value = '';
});


// Submit
dynamicForm.addEventListener('submit', (event) => {

  event.preventDefault();

  const formData = new FormData(dynamicForm);

  const data = {};

  for (const [key, value] of formData.entries()) {
    data[key] = value;
  }

  console.log(data);

  result.textContent = JSON.stringify(data, null, 2);
});


// Reset
resetBtn.addEventListener('click', () => {

  fieldsContainer.innerHTML = '';

  result.textContent = '';

  fieldCount = 0;

  inputType.value = '';
});
